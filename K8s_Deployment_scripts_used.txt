#To create a new service or pods:
kubectl apply -f {File path}


#To get description of the pods created:
kubectl describe pods {pod name} -n {Name-space name}


#To delete any service:
kubectl delete deployment workspace-service -n lovable-core


#To run data-base connection:
kubectl exec -it {Database used(to be find from the list of pods)} -n {Name-space name} -- psql -U postgres


#To do port forwading:
kubectl port-forward pod/<pod-name> <local-port>:<container-port> -n <namespace>
kubectl port-forward pod/pgvector-0 5432:5432 -n lovable-core
